  \nameparts{#1}% split the name
  % given name initials
  \ifblank{\namepartgiven}{}{%
    \textsc{\namepartgiveni}%
    \addperiod\addspace
  }%
  % family name small caps
  \textsc{\namepartfamily}%
  % separator logic: comma between, 'and' before last
  \ifthenelse{\value{listcount}<\numexpr\value{liststop}-1\relax}%
    {\addcomma\space}% between early names
    {\ifthenelse{\value{listcount}=\numexpr\value{liststop}-1\relax}%
      {\addcomma\space \textsc{and}\space}% before final author
      {}%
    }%
